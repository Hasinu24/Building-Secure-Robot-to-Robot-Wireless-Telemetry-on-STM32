/* ------------------------------------------------------------------
 * LED Helpers
 * ------------------------------------------------------------------ */
void LED_Blink_Send(void) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_SET);
    HAL_Delay(80);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_RESET);
}

void LED_Blink_Receive(void) {
    for (int i = 0; i < 3; i++) {
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_SET);
        HAL_Delay(200);
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_RESET);
        HAL_Delay(200);

    }
}
void LED_Blink_read(void) {
    for (int i = 0; i <3 ; i++) {
        HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_SET);
        HAL_Delay(600);


    }
}

/* ------------------------------------------------------------------
 * Zigbee_Receive - Support for 0x90 and 0x91 API Frames
 * ------------------------------------------------------------------ */
int Zigbee_Receive(uint8_t* outPayload, uint8_t* outLen)
{
    uint8_t byte, lenH, lenL;
    uint16_t frameLen;

    /* -----------------------------
     * STEP 1: Wait for start byte (0x7E)
     * ----------------------------- */
    if (HAL_UART_Receive(&huart1, &byte, 1, 10) != HAL_OK)
        return 0;

    if (byte != 0x7E)
        return 0; // Not a valid ZigBee frame

    /* -----------------------------
     * STEP 2: Read frame length (2 bytes)
     * ----------------------------- */
    if (HAL_UART_Receive(&huart1, &lenH, 1, 500) != HAL_OK)
        return 0;

    if (HAL_UART_Receive(&huart1, &lenL, 1, 500) != HAL_OK)
        return 0;

    frameLen = ((uint16_t)lenH << 8) | lenL;

    /* Safety check */
    if (frameLen < 1 || frameLen > 110)
        return 0;

    /* -----------------------------
     * STEP 3: Read full frame (data + checksum)
     * ----------------------------- */
    uint8_t frameBuf[120] = {0};

    for (uint16_t i = 0; i < frameLen + 1; i++)
    {
        if (HAL_UART_Receive(&huart1, &frameBuf[i], 1, 500) != HAL_OK)
            return 0;
    }

    /* -----------------------------
     * STEP 4: CHECK CHECKSUM (VERY IMPORTANT)
     * ----------------------------- */
    uint8_t sum = 0;

    for (uint16_t i = 0; i < frameLen; i++)
    {
        sum += frameBuf[i];  // add all frame data bytes
    }

    uint8_t receivedChecksum = frameBuf[frameLen];

    if ((sum + receivedChecksum) != 0xFF)
    {
        //  BAD DATA  ignore
        return 0;
    }
    /*  If correct, continue */

    /* -----------------------------
     * STEP 5: Identify frame type
     * ----------------------------- */
    uint8_t frameType = frameBuf[0];
    uint8_t offset = 0;

    if (frameType == 0x90)
        offset = 12;   // Normal receive frame
    else if (frameType == 0x91)
        offset = 18;   // Extended receive frame
    else
        return 0;      // Ignore other frames

    /* -----------------------------
     * STEP 6: Calculate payload size
     * ----------------------------- */
    int16_t pLen = frameLen - offset;

    if (pLen <= 0 || pLen > 80)
        return 0;

    /* -----------------------------
     * STEP 7: Extract ONLY your data
     * ----------------------------- */
    memcpy(outPayload, &frameBuf[offset], pLen);

    *outLen = (uint8_t)pLen;

    /* -----------------------------
     * STEP 8: SUCCESS
     * ----------------------------- */
    return 1;
}

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_USART1_UART_Init();
    MX_USART2_UART_Init();
    MX_I2C1_Init();

    int msgLen = snprintf(msgBuff, BUFF_SIZE, "\r\n=== BOARD %d STARTED ===\r\n", THIS_BOARD);
    HAL_UART_Transmit(&huart2, (uint8_t*)msgBuff, msgLen, 100);

    uint32_t lastSendTime = 0;

    while (1)
    {
        /* --- UART ERROR RECOVERY --- */
        if (__HAL_UART_GET_FLAG(&huart1, UART_FLAG_ORE)) {
            __HAL_UART_CLEAR_OREFLAG(&huart1);
        }
        if (__HAL_UART_GET_FLAG(&huart1, UART_FLAG_FE)) {
            __HAL_UART_CLEAR_FEFLAG(&huart1);
        }

        /* --- SEND BLOCK (Every 3 seconds) --- */
        if ((HAL_GetTick() - lastSendTime) >= 3000)
        {
            lastSendTime = HAL_GetTick();

            uint8_t rawData[2] = {(uint8_t)MY_ZONE, (uint8_t)temperature};
            uint8_t encryptedData[2];

            encryptedData[0] = rawData[0] ^ XOR_KEY;
            encryptedData[1] = rawData[1] ^ XOR_KEY;



            msgLen = snprintf(msgBuff, BUFF_SIZE, "\r\n[TX] Sending Zone:%d Temp:%dC (encrypted: %02X %02X)",
                              MY_ZONE, temperature, encryptedData[0], encryptedData[1]);
            HAL_UART_Transmit(&huart2, (uint8_t*)msgBuff, msgLen, 100);

            sendFrame(encryptedData, 2);
            LED_Blink_Send();

            if (++temperature > 100) temperature = START_TEMP;
        }

        /* --- RECEIVE BLOCK --- */
        uint8_t payload[81] = {0};
        uint8_t pLen = 0;

        if (Zigbee_Receive(payload, &pLen))
        {
            /* ---------- ACK CHECK FIRST ---------- */
            if (pLen == 1)
            {
                uint8_t ack = payload[0] ^ XOR_KEY;

                if (ack == 0xAC)
                {
                    msgLen = snprintf(msgBuff, BUFF_SIZE,
                        "\r\n[ACK] Received confirmation");
                    HAL_UART_Transmit(&huart2, (uint8_t*)msgBuff, msgLen, 100);
                }
            }

            /* ---------- NORMAL DATA ---------- */
            else if (pLen >= 2)
            {
                uint8_t zone = payload[0] ^ XOR_KEY;
                uint8_t temp = payload[1] ^ XOR_KEY;
                lastZone = zone;

                msgLen = snprintf(msgBuff, BUFF_SIZE,
                    "\r\n[RX] Zone:%d Temp:%dC", zone, temp);
                HAL_UART_Transmit(&huart2, (uint8_t*)msgBuff, msgLen, 100);

                //  SEND ACK BACK
                uint8_t ack = 0xAC ^ XOR_KEY;
                sendFrame(&ack, 1);


                #if THIS_BOARD == 2
                HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);
                EEPROM_Write(0, zone, &temp, 1);
                HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_SET);
                #endif

                LED_Blink_Receive();
            }
        }

        /* --- BUTTON READ (Board 2 Only) --- */
        #if THIS_BOARD == 2
        if (HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13) == GPIO_PIN_RESET)
        {
            uint8_t eepromVal = 0;
            EEPROM_Read(0, lastZone, &eepromVal, 1);
            LED_Blink_read();

            msgLen = snprintf(msgBuff, BUFF_SIZE, "\r\n[EEPROM] Zone:%d saved Temp:%d", lastZone, eepromVal);
            HAL_UART_Transmit(&huart2, (uint8_t*)msgBuff, msgLen, 100);
            HAL_Delay(500); // Debounce
        }
        #endif
    }
}

    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOC_CLK_ENABLE();
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* LED PA5 */
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_RESET);
    GPIO_InitStruct.Pin = GPIO_PIN_5;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

    /* Button PC13 */
    GPIO_InitStruct.Pin = GPIO_PIN_13;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(GPIOC, &GPIO_InitStruct);

    /* EEPROM Write Protect PB5 */

    GPIO_InitStruct.Pin = GPIO_PIN_5;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);
}
